# SAP COMMERCE (HYBRIS) IMPEX AUTOMATION STUDIO - PROJECT COMPLETION REPORT

================================================================================
1. PROCESS ANALYSIS & IDENTIFIED PAIN POINTS
================================================================================
Previously, creating localized ImpEx files for global e-commerce deployments was
a highly repetitive, manual, and error-prone workflow. When content managers or
developers pushed seasonal or welcome updates, they were forced to execute the
following manual steps:

* Content Replication: Manually copy-pasting base English text into external 
  translation services for each required target language (DE, ES, FR, IT, UK).
* Lowercase Formatting Enforcements: Reviewing and rewriting raw HTML elements
  (e.g., transforming <DIV CLASS="hero"> to <div class="hero">) to ensure strict 
  compliance with modern web and database standards without breaking inner text.
* Escape String Manipulation: Replacing single-double quotes ('') or unescaped 
  quotes with the exact double-double quotes ("") required by the Hybris data 
  importer. Failure to do so routinely caused cronjob validation crashes.
* File Assembly: Manually duplicating header configuration strings and building
  the layout line item arrays row by row in a text editor.

This structural fragmentation resulted in substantial overhead, averaging 15 to 
20 minutes per component update, and scaled exponentially with the volume of text.

================================================================================
2. RECOMMENDED AUTOMATION APPROACH & IMPLEMENTATION
================================================================================
The engineered solution implements a centralized Full-Stack Web Application 
Workspace (Angular 17+ Core Client connected to a Node.js/Express Engine) that
bypasses code-editor configurations entirely. This gives non-technical content 
managers and engineers equal ease of access.

The core architecture completely automates the workflow via:
* DeepL Free API Tier Integration: Securely routes bulk language parameters via
  asynchronous application transport wrappers directly to translation cells.
* Regular Expression Standardization Rules: Isolates HTML element blocks to 
  force-normalize markup tags to lowercase, maintain inner casing value strings, 
  and automatically format standalone microdata fields like itemscope=""itemscope"".
* String Grid Escaping Engine: Automatically encloses attribute layouts inside 
  escaped double-double quotes and binds matching dynamic regional codes (de-de, 
  es-es, fr-fr, it-it, en-gb) directly onto the final downloadable .impex files.

/**
 * Core Impex Standardization Engine
 * Formats HTML tags to lowercase, applies double-double quotes to attributes,
 * and standardizes language locales to specific regional formats (e.g., de-de, en-gb).
 */
exports.buildImpexMatrix = (headerConfig, uid, contentMap) => {
    const lines = [];
    const localeMap = {
        'en': 'en-gb',
        'de': 'de-de',
        'es': 'es-es',
        'fr': 'fr-fr',
        'it': 'it-it',
        'uk': 'en-gb'
    };
    
    for (const [lang, rawHtml] of Object.entries(contentMap)) {
        if (!rawHtml) continue;
        const targetLocale = localeMap[lang.toLowerCase()] || lang.toLowerCase();
        const dynamicHeader = headerConfig 
            ? headerConfig.replace('$lang', targetLocale)
            : `INSERT_UPDATE CMSParagraphComponent;uid[unique=true];content[lang=${targetLocale}]`;
            
        lines.push(dynamicHeader);
        let processedHtml = rawHtml;
        
        processedHtml = processedHtml.replace(/<[^>]+>/g, (tag) => {
            const isClosing = tag.startsWith('</');
            if (isClosing) return tag.toLowerCase();
            
            const tagMatch = tag.match(/^<([A-Z0-9]+)/i);
            if (!tagMatch) return tag;
            const tagName = tagMatch[1].toLowerCase();
            
            const attrPart = tag.substring(tagMatch[0].length, tag.length - 1);
            const attrRegex = /([A-Z0-9-_]+)(?:\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+)))?/gi;
            const attrs = [];
            let match;
            let hasItemscope = false;
            
            while ((match = attrRegex.exec(attrPart)) !== null) {
                const key = match[1].toLowerCase();
                const value = match[2] || match[3] || match[4] || "";
                if (key === 'itemscope') {
                    hasItemscope = true;
                    continue;
                }
                attrs.push(`${key}=""\${value}""`);
            }
            
            if (hasItemscope || tag.toLowerCase().includes('itemscope')) {
                attrs.unshift('itemscope=""itemscope""');
            }
            
            return attrs.length > 0 
                ? `<\${tagName} \${attrs.join(' ')}>`
                : `<\${tagName}>`;
        });
        
        lines.push(`;\${uid};"\${processedHtml}"`);
        lines.push("");
    }
    return lines.join('\n').trim();
};

const impexService = require('../services/impex.service');
const axios = require('axios');

exports.generateImpex = (req, res) => {
    try {
        const { headerConfig, uid, contentMap } = req.body;
        if (!uid || !contentMap || typeof contentMap !== 'object') {
            return res.status(400).json({ error: "Missing required parameters." });
        }
        const impexOutput = impexService.buildImpexMatrix(headerConfig, uid, contentMap);
        return res.status(200).json({ success: true, impexData: impexOutput });
    } catch (error) {
        console.error("[COMPILER ERROR]:", error);
        return res.status(500).json({ error: "Internal compilation failure" });
    }
};

exports.autoTranslateContent = async (req, res) => {
    try {
        console.log("\n[DEEPL TRANSLATION] Incoming payload request received...");
        const { englishText } = req.body;
        if (!englishText) {
            return res.status(400).json({ error: "No English base text provided." });
        }

        const apiKey = "5daf7c0d-e4d8-4e3e-b4b5-0595e166de16:fx";
        const endpoint = 'https://api-free.deepl.com/v2/translate';
        const targetLanguages = ['DE', 'ES', 'FR', 'IT', 'UKR'];
        const finalTranslations = { en: englishText };
        let authFailureDetected = null;

        await Promise.all(targetLanguages.map(async (lang) => {
            try {
                const params = new URLSearchParams();
                params.append('text', englishText);
                params.append('target_lang', lang);
                params.append('tag_handling', 'html');

                const response = await axios.post(endpoint, params.toString(), {
                    headers: {
                        'Authorization': `DeepL-Auth-Key \${apiKey}`,
                        'Content-Type': 'application/x-www-form-urlencoded'
                    }
                });

                const webLangKey = lang === 'UKR' ? 'uk' : lang.toLowerCase();
                if (response.data && response.data.translations && response.data.translations[0]) {
                    finalTranslations[webLangKey] = response.data.translations[0].text;
                } else {
                    finalTranslations[webLangKey] = englishText;
                }
            } catch (err) {
                if (err.response && err.response.status === 403) {
                    authFailureDetected = `DeepL API Key Rejected (403 Forbidden).`;
                }
                const webLangKey = lang === 'UKR' ? 'uk' : lang.toLowerCase();
                finalTranslations[webLangKey] = englishText;
            }
        }));

        if (authFailureDetected) {
            console.error(`[CRITICAL] \${authFailureDetected}`);
            return res.status(401).json({ error: authFailureDetected });
        }

        console.log("[SUCCESS] Translations processed cleanly.");
        return res.status(200).json({ success: true, translations: finalTranslations });
    } catch (error) {
        console.error("[SYSTEM EXCEPTION] Pipeline crashed:", error.message);
        return res.status(500).json({ error: `DeepL translation failed: \${error.message}` });
    }
};

import { Component } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { ImpexApiService, ImpexPayload } from '../../services/impex-api.service';

@Component({
  selector: 'app-impex-workspace',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './impex-workspace.component.html',
  styles: [`
    .workspace-container { padding: 20px; font-family: sans-serif; }
    .editor-grid { display: grid; grid-template-columns: 1fr 1fr; gap: 20px; margin-top: 15px; }
    .lang-card { border: 1px solid #ccc; padding: 10px; margin-bottom: 10px; border-radius: 4px; }
    textarea { width: 100%; font-family: monospace; }
    pre { background: #2d2d2d; color: #f1f1f1; padding: 15px; border-radius: 4px; overflow-x: auto; }
  `]
})
export class ImpexWorkspaceComponent {
  headerConfig: string = 'INSERT_UPDATE CMSParagraphComponent;uid[unique=true];content[lang=\$lang]';
  uid: string = 'HomepageWelcomeParagraph';

  languages: string[] = ['en', 'de', 'es', 'fr', 'it', 'uk'];
  contentMap: { [key: string]: string } = {
    en: '<DIV CLASS="hero">Welcome</DIV>',
    de: '',
    es: '',
    fr: '',
    it: '',
    uk: ''
  };

  outputResult: string = '';

  constructor(private apiService: ImpexApiService) { }
  
  downloadImpexFile() {
    if (!this.outputResult) return;
    const blob = new Blob([this.outputResult], { type: 'text/plain;charset=utf-8;' });
    const link = document.createElement('a');
    const filename = `\${this.uid || 'content_update'}_\${new Date().toISOString().slice(0, 10)}.impex`;
    const url = URL.createObjectURL(blob);

    link.setAttribute('href', url);
    link.setAttribute('download', filename);
    link.style.visibility = 'hidden';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  }

  processImpex() {
    const payload: ImpexPayload = {
      headerConfig: this.headerConfig,
      uid: this.uid,
      contentMap: this.contentMap
    };
    this.apiService.generateImpex(payload).subscribe({
      next: (res: any) => {
        this.outputResult = res.impexData;
      },
      error: (err) => {
        this.outputResult = 'Error generating impex data content.';
      }
    });
  }

  autoPopulateTranslations() {
    const englishSource = this.contentMap['en'];
    if (!englishSource) {
      alert('Please provide an English base content string first!');
      return;
    }
    this.apiService.autoTranslate(englishSource).subscribe({
      next: (res: any) => {
        if (res && res.success) {
          this.contentMap = { ...this.contentMap, ...res.translations };
        }
      },
      error: (err) => {
        const serverErrorMessage = err.error?.error || err.message || 'Unknown network error';
        alert(`Translation Error:\n\${serverErrorMessage}`);
      }
    });
  }

  copyToClipboard() {
    if (!this.outputResult) return;
    navigator.clipboard.writeText(this.outputResult);
    alert('Impex copied cleanly!');
  }
}
